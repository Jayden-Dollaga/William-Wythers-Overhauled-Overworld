=== STEP 5 — CHECK WHAT FILES WERE DELETED vs KEPT ===

  python3 -c "
  import os
  orig = set()
  alpha = set()
  for root,dirs,files in os.walk('WWOO_ORIGINAL/data'):
      for f in files:
          if f.endswith('.json'):
              orig.add(os.path.relpath(os.path.join(root,f), 'WWOO_ORIGINAL/data'))
  for root,dirs,files in os.walk('wwoo-26.2-port-fixed/data'):
      for f in files:
          if f.endswith('.json'):
              alpha.add(os.path.relpath(os.path.join(root,f), 'wwoo-26.2-port-fixed/data'))
  deleted = orig - alpha
  added = alpha - orig
  print(f'Deleted in 26.2 alpha: {len(deleted)}')
  for f in sorted(deleted)[:30]: print(f'  {f}')
  print(f'New in 26.2 alpha: {len(added)}')
  for f in sorted(added)[:30]: print(f'  {f}')
  "
